        function prepararAcordeon() {
            var toggles = panel.querySelectorAll('.menu_navegacion_toggle, .menu_navegacion_subtoggle');
            for (var t = 0; t < toggles.length; t++) {
                var li = toggles[t].parentElement;
                var tieneSub = false;
                for (var i = 0; i < li.children.length; i++) {
                    if (li.children[i].tagName === 'UL') {
                        tieneSub = true;
                        break;
                    }
                }
                if (tieneSub && toggles[t].getAttribute('onclick')) {
                    toggles[t].removeAttribute('onclick');
                }
            }
        }

        function abrirMenu() {
            document.body.classList.add('menu-mobile-abierto');
            btn.setAttribute('aria-expanded', 'true');
            prepararAcordeon();
        }

        function cerrarMenu() {
            document.body.classList.remove('menu-mobile-abierto');
            btn.setAttribute('aria-expanded', 'false');
        }

        btn.addEventListener('click', function () {
            if (document.body.classList.contains('menu-mobile-abierto')) {
                cerrarMenu();
            } else {
                abrirMenu();
            }
        });
        if (cerrar) {
            cerrar.addEventListener('click', cerrarMenu);
        }

        panel.addEventListener('click', function (e) {
            if (!esMobile()) return;
            var toggle = e.target.closest('.menu_navegacion_toggle, .menu_navegacion_subtoggle');
            if (!toggle) return;
            var li = toggle.parentElement;
            var submenu = null;
            for (var i = 0; i < li.children.length; i++) {
                if (li.children[i].tagName === 'UL') {
                    submenu = li.children[i];
                    break;
                }
            }
            if (!submenu) return;
            e.preventDefault();
            e.stopPropagation();
            li.classList.toggle('is-open');

    })();
